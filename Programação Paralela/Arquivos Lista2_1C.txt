//Todos os processos enviam um escalar ao processo zero
#include <mpi.h>
#include <stdio.h>

int main(int argc, char *argv[])
{
    int p, id, tag = 1, destino, numero, origem;
    MPI_Status status;

    MPI_Init(&argc, &argv);

    MPI_Comm_rank(MPI_COMM_WORLD, &id);
    MPI_Comm_size(MPI_COMM_WORLD, &p);

    if (id != 0)
    {
        destino = 0;
        numero = id * 100;

        MPI_Send(&numero, 1, MPI_INT, destino, tag, MPI_COMM_WORLD);
    }
    else
    {
        for (origem = 1; origem < p; origem++){
            MPI_Recv(&numero,1,MPI_INT,origem,tag,MPI_COMM_WORLD,&status);
            printf("Processo 0 recebeu %d do Processo %d\n", numero, origem);
        }
    }

    MPI_Finalize();

    return 0;
}
